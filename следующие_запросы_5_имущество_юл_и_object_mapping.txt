113. Join: договоры ИМЩ ЮЛ -> убытки
====================================

Сохрани как:
113_join_contracts_losses_imush_ul_sample.csv

WITH c AS (
    SELECT
        CONTRACT_ID,
        CLIENT_ID,
        OBJECT_ID,
        PRODUCT_GROUP,
        PRODUCT_TYPE,
        PRODUCT_1C_NAME,
        PREM_SUM_RUB,
        OTV_SUM_RUB
    FROM DM_RISK_AVATAR.CONTRACTS
    WHERE (
           LOWER(PRODUCT_GROUP) LIKE '%страхование юл%'
        OR LOWER(PRODUCT_GROUP) LIKE '%залоговое страхование юл%'
        OR LOWER(PRODUCT_TYPE) LIKE '%имущество юл%'
        OR LOWER(PRODUCT_TYPE) LIKE '%имуществ%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%имущество юридических лиц%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%июл%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%залоги%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%сверхлимит%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%смр%'
        OR LOWER(PRODUCT_1C_NAME) LIKE '%строительно-монтаж%'
    )
      AND ROWNUM <= 100000
),
l AS (
    SELECT
        CONTRACT_ID,
        CLAIM_ID,
        PAY_CLAIM_TOTAL
    FROM DM_RISK_AVATAR.LOSSES
)
SELECT
    COUNT(*) AS joined_rows,
    COUNT(DISTINCT c.CONTRACT_ID) AS sample_contracts,
    COUNT(DISTINCT l.CLAIM_ID) AS matched_claims,
    COUNT(l.CONTRACT_ID) AS matched_loss_rows,
    SUM(l.PAY_CLAIM_TOTAL) AS matched_pay_claim_total
FROM c
LEFT JOIN l
  ON c.CONTRACT_ID = l.CONTRACT_ID;

WITH s AS (
    SELECT OBJ_ADDRESS
    FROM DM_RISK_AVATAR.STG_CONTRACT_OBJRISK
    WHERE OBJ_ADDRESS IS NOT NULL
      AND (
           LOWER(OBJ_TYPE) LIKE '%юридических лиц%'
        OR LOWER(OBJ_TYPE) LIKE '%имущество%'
        OR LOWER(OBJ_TYPE) LIKE '%строен%'
        OR LOWER(OBJ_TYPE) LIKE '%здан%'
        OR LOWER(OBJ_TYPE) LIKE '%склад%'
        OR LOWER(RISK_NAME) LIKE '%июл%'
        OR LOWER(RISK_NAME) LIKE '%имущество%'
        OR LOWER(RISK_NAME) LIKE '%залоги%'
        OR LOWER(RISK_NAME) LIKE '%смр%'
        OR LOWER(RISK_NAME) LIKE '%строительно-монтаж%'
      )
      AND ROWNUM <= 10000
)
SELECT
    COUNT(*) AS sample_stg_rows,
    COUNT(b.CAD_IND) AS matched_buildings_rows
FROM s
LEFT JOIN DM_RISK_AVATAR.BUILDINGS b
  ON UPPER(TRIM(s.OBJ_ADDRESS)) = UPPER(TRIM(b.ADDRESS));
