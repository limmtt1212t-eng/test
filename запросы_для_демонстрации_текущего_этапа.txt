/*
ЗАПРОСЫ ДЛЯ ДЕМОНСТРАЦИИ ТЕКУЩЕГО ЭТАПА
========================================

Как запускать:
1. Открыть этот файл в DBeaver.
2. Выделять и запускать по одному запросу.
3. Показывать результаты на экране без выгрузки конфиденциальных строк.

Что было получено на момент исходной выгрузки:
- запрос 1 из переданного файла: выгружено 5000 строк по лимиту;
- запрос 2 из переданного файла: полный результат 2446 строк;
- запрос 3 из переданного файла: полный результат 66 строк.

Важно:
- первые два исходных запроса не отсортированы по дате;
- строка результата не всегда равна уникальному объекту или договору;
- operational_archive является статусом задачи, а не доказательством
  юридически заключённого договора;
- проверка ФИАС ниже относится к выбранной технической популяции,
  а не автоматически ко всей базе.
*/


/*
ЗАПРОС 1. ОБЪЁМ РЕЗУЛЬТАТА ТРЁХ ПЕРЕДАННЫХ SQL
-----------------------------------------------

Что показывает:
- сколько строк воспроизводится по каждому исходному запросу
  при верхнем ограничении 5000;
- достигнут ли LIMIT или запрос вернул полный результат меньшего объёма.

Результат обезличен: одна строка на каждый исходный SQL.
*/

with
query_1_rows as (
    select 1
    from base_insurance_object obj
    join base_insurance_object_characteristics oc
        on obj.id = oc.insurance_object_id
    join base_insurance_object_conditions ot
        on oc.id = ot.characteristics_id
    join base_geo_address geo
        on geo.id = obj.geo_address_id
    where obj.elementary_obj_type in (
        'nedv_ul_and_ip',
        'business_interruption'
    )
    limit 5000
),
query_2_rows as (
    select 1
    from bps_request_ins_task t
    join bps_request_ins r
        on r.id = t.request_ins_id
    join bps_request_ins_task_insurance_object tobj
        on tobj.parent_id = t.id
    join base_insurance_object_characteristics op
        on op.id = tobj.characteristics_id
    join base_insurance_object obj
        on obj.id = op.insurance_object_id
    join base_insurance_object_conditions ot
        on op.id = ot.characteristics_id
    join base_geo_address geo
        on geo.id = obj.geo_address_id
    join bps_corporate_crm bcc
        on bcc.id = r.corporate_crm_id
    join bps_contractor bc
        on bc.id = bcc.client_id
    where t.task_type = 'draft_contract'
      and t.status = 'operational_archive'
      and (
          t.ins_document_type = 'new_ins_contract'
          or t.ins_document_type = 'ins_contract_prolong'
          or t.ins_document_type is null
      )
      and obj.elementary_obj_type in (
          'nedv_ul_and_ip',
          'business_interruption'
      )
    limit 5000
),
query_3_rows as (
    select 1
    from bps_corporate_claim cl
    join bps_corporate_crm bcc
        on bcc.id = cl.appliant_id
    join bps_contractor bc
        on bc.id = bcc.client_id
    order by cl.d_create desc
    limit 5000
),
result_counts as (
    select
        1 as query_number,
        'Недвижимость и BI'::text as query_name,
        count(*) as exported_row_count
    from query_1_rows

    union all

    select
        2,
        'Объекты в связке с оформлением договоров',
        count(*)
    from query_2_rows

    union all

    select
        3,
        'Убытки по переданной связи заявителя',
        count(*)
    from query_3_rows
)
select
    query_number
        as "Номер запроса",
    query_name
        as "Название",
    exported_row_count
        as "Строк в воспроизведённой выгрузке",
    case
        when exported_row_count = 5000
            then 'Достигнут LIMIT: полный объём этим запуском не определён'
        else 'Полный результат меньше установленного LIMIT'
    end
        as "Комментарий"
from result_counts
order by query_number;


/*
ЗАПРОС 2. СВОДКА ПО ОБЪЕКТНОЙ ОСНОВЕ ПЕРВОЙ МОДЕЛИ
---------------------------------------------------

Техническая единица:
один объект недвижимости в одной выбранной задаче договора.

Что показывает:
- бывают ли несколько объектов в одном договоре;
- бывают ли несколько объектов по одному адресу;
- сколько объектов без адреса;
- заполненность ИНН, ФИАС, адреса, площади, координат и стоимости;
- сколько объектов пока нельзя уверенно сопоставлять с ЕГРН.

Берётся не более 5000 наиболее свежих подходящих договоров.
Результат обезличен и состоит из одной строки.
*/

with task_candidates as (
    select
        c.id as contract_id,
        c.contractor_id,
        r.corporate_crm_id,
        t.id as task_id,
        coalesce(
            t.d_conclusion_ins_contract::timestamp with time zone,
            c.d_sign_contract,
            t.d_create
        ) as as_of_date,
        row_number() over (
            partition by c.id
            order by
                t.d_conclusion_ins_contract desc nulls last,
                t.d_create desc nulls last,
                t.d_change desc nulls last,
                t.id desc
        ) as task_rank
    from bps_request_ins_task t
    join bps_request_ins r
        on r.id = t.request_ins_id
    join bps_contract c
        on c.id = r.contract_id
    where t.task_type = 'draft_contract'
      and t.status = 'operational_archive'
      and (
          t.ins_document_type = 'new_ins_contract'
          or t.ins_document_type = 'ins_contract_prolong'
          or t.ins_document_type is null
      )
      and t.d_delete is null
      and r.d_delete is null
      and c.d_delete is null
      and t.ins_refuse is not true
),
selected_tasks as (
    select *
    from task_candidates
    where task_rank = 1
),
property_tasks as (
    select task.*
    from selected_tasks task
    where exists (
        select 1
        from bps_request_ins_task_insurance_object tobj_check
        join base_insurance_object_characteristics ch_check
            on ch_check.id = tobj_check.characteristics_id
        join base_insurance_object obj_check
            on obj_check.id = ch_check.insurance_object_id
        where tobj_check.parent_id = task.task_id
          and obj_check.elementary_obj_type = 'nedv_ul_and_ip'
          and obj_check.d_delete is null
    )
),
sampled_tasks as (
    select *
    from property_tasks
    order by
        as_of_date desc nulls last,
        contract_id desc
    limit 5000
),
object_candidates as (
    select
        task.contract_id,
        task.task_id,
        policyholder.inn as policyholder_inn,
        tobj.id as task_object_link_id,
        obj.id as object_id,
        obj.description as object_description,
        obj.original_address,
        geo.full_address,
        geo.fias_code,
        geo.latitude,
        geo.longitude,
        ch.insurance_value,
        ch.characteristics,
        coalesce(
            nullif(btrim(geo.full_address), ''),
            nullif(btrim(obj.original_address), '')
        ) as address_for_analysis,
        row_number() over (
            partition by task.task_id, obj.id
            order by
                tobj.d_change desc nulls last,
                tobj.d_create desc nulls last,
                ch.version_start_date desc nulls last,
                ch.version_number desc nulls last,
                tobj.id desc,
                ch.id desc
        ) as object_rank
    from sampled_tasks task
    join bps_request_ins_task_insurance_object tobj
        on tobj.parent_id = task.task_id
    join base_insurance_object_characteristics ch
        on ch.id = tobj.characteristics_id
    join base_insurance_object obj
        on obj.id = ch.insurance_object_id
    left join base_geo_address geo
        on geo.id = obj.geo_address_id
    left join bps_contractor policyholder
        on policyholder.id = task.contractor_id
    where obj.elementary_obj_type = 'nedv_ul_and_ip'
      and obj.d_delete is null
),
object_rows as (
    select *
    from object_candidates
    where object_rank = 1
),
address_statistics as (
    select
        contract_id,
        address_for_analysis,
        count(*) as objects_at_address
    from object_rows
    where address_for_analysis is not null
    group by
        contract_id,
        address_for_analysis
),
contract_statistics as (
    select
        contract_id,
        count(*) as object_count,
        count(distinct address_for_analysis) as address_count,
        count(*) filter (
            where address_for_analysis is null
        ) as objects_without_address
    from object_rows
    group by contract_id
),
contract_statistics_final as (
    select
        contract.contract_id,
        contract.object_count,
        contract.address_count,
        contract.objects_without_address,
        coalesce(max(address.objects_at_address), 0)
            as max_objects_at_same_address
    from contract_statistics contract
    left join address_statistics address
        on address.contract_id = contract.contract_id
    group by
        contract.contract_id,
        contract.object_count,
        contract.address_count,
        contract.objects_without_address
),
contract_summary as (
    select
        count(*) as contract_count,
        count(*) filter (
            where object_count = 1
        ) as single_object_contract_count,
        count(*) filter (
            where object_count > 1
        ) as multi_object_contract_count,
        count(*) filter (
            where max_objects_at_same_address > 1
        ) as contracts_with_shared_address,
        count(*) filter (
            where objects_without_address > 0
        ) as contracts_with_missing_address,
        max(object_count) as maximum_objects_in_contract
    from contract_statistics_final
),
object_summary as (
    select
        count(*) as object_count,
        count(*) filter (
            where nullif(
                regexp_replace(
                    coalesce(policyholder_inn, ''),
                    '[^0-9]',
                    '',
                    'g'
                ),
                ''
            ) is not null
        ) as objects_with_inn,
        count(*) filter (
            where nullif(btrim(fias_code), '') is not null
        ) as objects_with_fias,
        count(*) filter (
            where address_for_analysis is not null
        ) as objects_with_address,
        count(*) filter (
            where latitude is not null
              and longitude is not null
        ) as objects_with_coordinates,
        count(*) filter (
            where characteristics is not null
              and characteristics <> '{}'::jsonb
              and characteristics <> '[]'::jsonb
              and characteristics <> 'null'::jsonb
        ) as objects_with_json,
        count(*) filter (
            where nullif(
                btrim(characteristics ->> 'total_area_sq_m'),
                ''
            ) is not null
        ) as objects_with_area,
        count(*) filter (
            where insurance_value is not null
        ) as objects_with_insurance_value,
        count(*) filter (
            where nullif(btrim(fias_code), '') is not null
              and nullif(
                  btrim(characteristics ->> 'total_area_sq_m'),
                  ''
              ) is not null
              and nullif(btrim(object_description), '') is not null
        ) as egrn_candidates_by_fias,
        count(*) filter (
            where address_for_analysis is not null
              and nullif(
                  btrim(characteristics ->> 'total_area_sq_m'),
                  ''
              ) is not null
              and nullif(btrim(object_description), '') is not null
        ) as egrn_candidates_by_address,
        count(*) filter (
            where not (
                (
                    nullif(btrim(fias_code), '') is not null
                    and nullif(
                        btrim(characteristics ->> 'total_area_sq_m'),
                        ''
                    ) is not null
                    and nullif(btrim(object_description), '') is not null
                )
                or
                (
                    address_for_analysis is not null
                    and nullif(
                        btrim(characteristics ->> 'total_area_sq_m'),
                        ''
                    ) is not null
                    and nullif(btrim(object_description), '') is not null
                )
            )
        ) as objects_without_egrn_key
    from object_rows
)
select
    contract.contract_count
        as "Договоров в технической выборке",
    object.object_count
        as "Объектов в технической выборке",
    contract.single_object_contract_count
        as "Договоров с одним объектом",
    contract.multi_object_contract_count
        as "Договоров с несколькими объектами",
    contract.contracts_with_shared_address
        as "Договоров с несколькими объектами по одному адресу",
    contract.contracts_with_missing_address
        as "Договоров с объектами без адреса",
    contract.maximum_objects_in_contract
        as "Максимум объектов в одном договоре",
    object.objects_with_inn
        as "Объектов с ИНН страхователя",
    round(
        100.0 * object.objects_with_inn
        / nullif(object.object_count, 0),
        1
    ) as "ИНН процент",
    object.objects_with_fias
        as "Объектов с ФИАС",
    round(
        100.0 * object.objects_with_fias
        / nullif(object.object_count, 0),
        1
    ) as "ФИАС процент",
    object.objects_with_address
        as "Объектов с адресом",
    round(
        100.0 * object.objects_with_address
        / nullif(object.object_count, 0),
        1
    ) as "Адрес процент",
    object.objects_with_coordinates
        as "Объектов с координатами",
    round(
        100.0 * object.objects_with_coordinates
        / nullif(object.object_count, 0),
        1
    ) as "Координаты процент",
    object.objects_with_json
        as "Объектов с непустым JSON",
    round(
        100.0 * object.objects_with_json
        / nullif(object.object_count, 0),
        1
    ) as "JSON процент",
    object.objects_with_area
        as "Объектов с площадью в JSON",
    round(
        100.0 * object.objects_with_area
        / nullif(object.object_count, 0),
        1
    ) as "Площадь процент",
    object.objects_with_insurance_value
        as "Объектов со страховой стоимостью",
    round(
        100.0 * object.objects_with_insurance_value
        / nullif(object.object_count, 0),
        1
    ) as "Страховая стоимость процент",
    object.egrn_candidates_by_fias
        as "Кандидатов ЕГРН по ФИАС площади и типу",
    object.egrn_candidates_by_address
        as "Кандидатов ЕГРН по адресу площади и типу",
    object.objects_without_egrn_key
        as "Объектов без полного рабочего набора для ЕГРН"
from contract_summary contract
cross join object_summary object;


/*
ЗАПРОС 3. ОБЕЗЛИЧЕННЫЙ ПРИМЕР ТЕКУЩЕЙ ОБЪЕКТНОЙ ТАБЛИЦЫ
--------------------------------------------------------

Что показывает:
- одна строка соответствует одному объекту выбранной задачи договора;
- объекты одного договора выводятся рядом;
- одинаковый условный номер адреса означает один адрес внутри договора;
- реальные ID, номера договоров, адреса, ИНН и суммы не выводятся.

Берётся до 20 наиболее свежих подходящих договоров.
*/

with task_candidates as (
    select
        c.id as contract_id,
        c.contractor_id,
        r.corporate_crm_id,
        t.id as task_id,
        t.industry as task_industry,
        coalesce(
            t.d_conclusion_ins_contract::timestamp with time zone,
            c.d_sign_contract,
            t.d_create
        ) as as_of_date,
        row_number() over (
            partition by c.id
            order by
                t.d_conclusion_ins_contract desc nulls last,
                t.d_create desc nulls last,
                t.d_change desc nulls last,
                t.id desc
        ) as task_rank
    from bps_request_ins_task t
    join bps_request_ins r
        on r.id = t.request_ins_id
    join bps_contract c
        on c.id = r.contract_id
    where t.task_type = 'draft_contract'
      and t.status = 'operational_archive'
      and (
          t.ins_document_type = 'new_ins_contract'
          or t.ins_document_type = 'ins_contract_prolong'
          or t.ins_document_type is null
      )
      and t.d_delete is null
      and r.d_delete is null
      and c.d_delete is null
      and t.ins_refuse is not true
),
property_tasks as (
    select *
    from task_candidates task
    where task_rank = 1
      and exists (
          select 1
          from bps_request_ins_task_insurance_object tobj_check
          join base_insurance_object_characteristics ch_check
              on ch_check.id = tobj_check.characteristics_id
          join base_insurance_object obj_check
              on obj_check.id = ch_check.insurance_object_id
          where tobj_check.parent_id = task.task_id
            and obj_check.elementary_obj_type = 'nedv_ul_and_ip'
            and obj_check.d_delete is null
      )
),
sampled_tasks as (
    select
        property_tasks.*,
        row_number() over (
            order by
                as_of_date desc nulls last,
                contract_id desc
        ) as example_contract_number
    from property_tasks
    order by
        as_of_date desc nulls last,
        contract_id desc
    limit 20
),
object_candidates as (
    select
        task.example_contract_number,
        task.contract_id,
        task.task_id,
        task.task_industry,
        policyholder.inn as policyholder_inn,
        crm.industry as crm_industry,
        obj.id as object_id,
        obj.description as object_description,
        obj.obj_type,
        obj.original_address,
        geo.full_address,
        geo.fias_code,
        ch.insurance_value,
        ch.is_pledged,
        ch.characteristics,
        coalesce(
            nullif(btrim(geo.full_address), ''),
            nullif(btrim(obj.original_address), '')
        ) as address_for_analysis,
        row_number() over (
            partition by task.task_id, obj.id
            order by
                tobj.d_change desc nulls last,
                tobj.d_create desc nulls last,
                ch.version_start_date desc nulls last,
                ch.version_number desc nulls last,
                tobj.id desc,
                ch.id desc
        ) as object_rank
    from sampled_tasks task
    join bps_request_ins_task_insurance_object tobj
        on tobj.parent_id = task.task_id
    join base_insurance_object_characteristics ch
        on ch.id = tobj.characteristics_id
    join base_insurance_object obj
        on obj.id = ch.insurance_object_id
    left join base_geo_address geo
        on geo.id = obj.geo_address_id
    left join bps_contractor policyholder
        on policyholder.id = task.contractor_id
    left join bps_corporate_crm crm
        on crm.id = task.corporate_crm_id
    where obj.elementary_obj_type = 'nedv_ul_and_ip'
      and obj.d_delete is null
),
object_rows as (
    select *
    from object_candidates
    where object_rank = 1
),
display_rows as (
    select
        object_rows.*,
        count(*) over (
            partition by contract_id
        ) as object_count,
        row_number() over (
            partition by contract_id
            order by object_id
        ) as object_number,
        case
            when address_for_analysis is null
                then null
            else dense_rank() over (
                partition by contract_id
                order by address_for_analysis
            )
        end as conditional_address_number,
        case
            when address_for_analysis is null
                then 0
            else count(*) over (
                partition by contract_id, address_for_analysis
            )
        end as objects_at_same_address
    from object_rows
)
select
    example_contract_number
        as "Условный номер договора",
    object_count
        as "Объектов в договоре",
    object_number
        as "Номер объекта в договоре",
    object_description
        as "Общее описание объекта",
    obj_type
        as "Тип объекта из системы",
    case
        when address_for_analysis is null
            then 'Нет'
        else 'Да'
    end as "Адрес заполнен",
    conditional_address_number
        as "Условный номер адреса внутри договора",
    objects_at_same_address
        as "Объектов по этому адресу",
    case
        when nullif(btrim(fias_code), '') is not null
            then 'Да'
        else 'Нет'
    end as "ФИАС заполнен",
    characteristics ->> 'total_area_sq_m'
        as "Площадь из JSON",
    characteristics ->> 'construction_year'
        as "Год постройки из JSON",
    case
        when characteristics is not null
          and characteristics <> '{}'::jsonb
          and characteristics <> '[]'::jsonb
          and characteristics <> 'null'::jsonb
            then 'Да'
        else 'Нет'
    end as "JSON характеристик заполнен",
    case
        when insurance_value is not null
            then 'Да'
        else 'Нет'
    end as "Страховая стоимость заполнена",
    case
        when is_pledged is true
            then 'Да'
        when is_pledged is false
            then 'Нет'
        else 'Не указано'
    end as "Есть залог",
    crm_industry
        as "Отрасль из CRM",
    task_industry
        as "Отрасль из задачи",
    case
        when nullif(
            regexp_replace(
                coalesce(policyholder_inn, ''),
                '[^0-9]',
                '',
                'g'
            ),
            ''
        ) is not null
            then 'Да'
        else 'Нет'
    end as "ИНН для СПАРК заполнен"
from display_rows
order by
    example_contract_number,
    object_number;
