1. Сохранить ключи и связи
==========================

Сохрани результат как:
анализ таблиц/results/04_ключи_и_связи.csv

SELECT
    c.owner,
    c.table_name,
    c.constraint_name,
    c.constraint_type,
    cc.position,
    cc.column_name,
    c.r_owner,
    c.r_constraint_name
FROM all_constraints c
JOIN all_cons_columns cc
  ON c.owner = cc.owner
 AND c.constraint_name = cc.constraint_name
WHERE c.owner = UPPER('DM_RISK_AVATAR')
  AND c.table_name IN (
      UPPER('DICT_INS_POTENTIAL_PRIORITY_RISKS'),
      UPPER('DICT_INS_POTENTIAL_RISK_PACK_ADD_FLAG'),
      UPPER('DICT_INS_POTENTIAL_RISK_PACK_ADD_FLAG_CONTRACT'),
      UPPER('STG_CONTRACT_OBJRISK')
  )
  AND c.constraint_type IN ('P', 'R', 'U')
ORDER BY c.table_name, c.constraint_type, c.constraint_name, cc.position;

==================================

    MIN(CONTRACT_END_DATE) AS min_contract_end_date,
    MAX(CONTRACT_END_DATE) AS max_contract_end_date,
    MIN(UPDATE_DTTM) AS min_update_dttm,
    MAX(UPDATE_DTTM) AS max_update_dttm
FROM DM_RISK_AVATAR.STG_CONTRACT_OBJRISK;

5. Поиск таблиц и колонок с убытками / claims / losses
======================================================

Это самый важный блок для скоринга и убыточности.
Сохрани результат как:
анализ таблиц/results/20_поиск_убытков_в_схеме.csv

SELECT
    c.owner,
    c.table_name,
    c.column_name,
    c.data_type,
    cc.comments
FROM all_tab_columns c
LEFT JOIN all_col_comments cc
  ON c.owner = cc.owner
 AND c.table_name = cc.table_name
 AND c.column_name = cc.column_name
WHERE c.owner = UPPER('DM_RISK_AVATAR')
  AND (
      LOWER(c.table_name) LIKE '%claim%'
      OR LOWER(c.table_name) LIKE '%loss%'
      OR LOWER(c.table_name) LIKE '%damage%'
      OR LOWER(c.table_name) LIKE '%убыт%'
      OR LOWER(c.table_name) LIKE '%ущерб%'
      OR LOWER(c.column_name) LIKE '%claim%'
      OR LOWER(c.column_name) LIKE '%loss%'
      OR LOWER(c.column_name) LIKE '%damage%'
      OR LOWER(c.column_name) LIKE '%убыт%'
      OR LOWER(c.column_name) LIKE '%ущерб%'
      OR LOWER(cc.comments) LIKE '%убыт%'
      OR LOWER(cc.comments) LIKE '%ущерб%'
      OR LOWER(cc.comments) LIKE '%выплат%'
  )
ORDER BY c.table_name, c.column_id;

10. Чего пока не делать
=======================

- SELECT *;
- полный COUNT, если он уже тормозит;
- GROUP BY по всей таблице без фильтра;
- COUNT DISTINCT по всей таблице без необходимости;
- join с другими большими таблицами без фильтра по датам.

Эта таблица слишком большая, сначала надо понять индексы и связи.
