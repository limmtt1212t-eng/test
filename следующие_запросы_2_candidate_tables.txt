3. Полная структура кандидатных таблиц
======================================

Сохрани результат как:
анализ таблиц/results/43_candidate_tables_columns.csv

SELECT
    c.owner,
    c.table_name,
    c.column_id,
    c.column_name,
    c.data_type,
    c.data_length,
    c.data_precision,
    c.data_scale,
    c.nullable,
    cc.comments
FROM all_tab_columns c
LEFT JOIN all_col_comments cc
  ON c.owner = cc.owner
 AND c.table_name = cc.table_name
 AND c.column_name = cc.column_name
WHERE c.owner = UPPER('DM_RISK_AVATAR')
  AND c.table_name IN (
      'LOSSES',
      'LOSSES_OBJECT_PAY',
      'CONTRACTS',
      'CONTRACT_DOPS',
      'BUILDINGS',
      'ESTATE_OBJECTS',
      'EGRN_CAD_OBJECT_PROCESSED',
      'EGRN_DATA',
      'EGRN_HASH',
      'LNK_CLIENT_OBJECT_CONTRACT',
      'LNK_CLIENT_CONTRACT'
  )
ORDER BY c.table_name, c.column_id;


4. Ключи и связи кандидатных таблиц
===================================

Сохрани результат как:
анализ таблиц/results/44_candidate_tables_constraints.csv

SELECT
    c.owner,
    c.table_name,
    c.constraint_name,
    c.constraint_type,
    cc.position,
    cc.column_name,
    c.r_owner,
    c.r_constraint_name
FROM all_constraints c
JOIN all_cons_columns cc
  ON c.owner = cc.owner
 AND c.constraint_name = cc.constraint_name
WHERE c.owner = UPPER('DM_RISK_AVATAR')
  AND c.table_name IN (
      'LOSSES',
      'LOSSES_OBJECT_PAY',
      'CONTRACTS',
      'CONTRACT_DOPS',
      'BUILDINGS',
      'ESTATE_OBJECTS',
      'EGRN_CAD_OBJECT_PROCESSED',
      'EGRN_DATA',
      'EGRN_HASH',
      'LNK_CLIENT_OBJECT_CONTRACT',
      'LNK_CLIENT_CONTRACT'
  )
  AND c.constraint_type IN ('P', 'R', 'U')
ORDER BY c.table_name, c.constraint_type, c.constraint_name, cc.position;

6. Date/timestamp колонки кандидатных таблиц
============================================

Сохрани результат как:
анализ таблиц/results/46_candidate_tables_date_columns.csv

SELECT
    table_name,
    column_name,
    data_type
FROM all_tab_columns
WHERE owner = UPPER('DM_RISK_AVATAR')
  AND table_name IN (
      'LOSSES',
      'LOSSES_OBJECT_PAY',
      'CONTRACTS',
      'CONTRACT_DOPS',
      'BUILDINGS',
      'ESTATE_OBJECTS',
      'EGRN_CAD_OBJECT_PROCESSED',
      'EGRN_DATA',
      'EGRN_HASH',
      'LNK_CLIENT_OBJECT_CONTRACT',
      'LNK_CLIENT_CONTRACT'
  )
  AND (
      data_type = 'DATE'
      OR data_type LIKE 'TIMESTAMP%'
  )
ORDER BY table_name, column_name;


7. Минимальная проверка связи LOSSES -> CONTRACTS
=================================================

Сохрани результат как:
анализ таблиц/results/47_join_losses_contracts_sample.csv

Важно:
это sample/rownum-проверка, не полная аналитика.

SELECT
    COUNT(*) AS sample_losses_rows,
    COUNT(c.CONTRACT_ID) AS matched_contracts_rows
FROM (
    SELECT CONTRACT_ID
    FROM DM_RISK_AVATAR.LOSSES
    WHERE CONTRACT_ID IS NOT NULL
      AND ROWNUM <= 10000
) l
LEFT JOIN DM_RISK_AVATAR.CONTRACTS c
  ON l.CONTRACT_ID = c.CONTRACT_ID;

    PRODUCT_GROUP,
    PRODUCT_TYPE,
    COUNT(*) AS row_count
FROM (
    SELECT PRODUCT_GROUP, PRODUCT_TYPE
    FROM DM_RISK_AVATAR.LOSSES
    WHERE ROWNUM <= 100000
