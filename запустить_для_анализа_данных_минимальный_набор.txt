91_contracts_quality_sample.csv

SELECT
    COUNT(*) AS sample_rows,
    MIN(LIABILITY_START_DATE) AS min_liability_start_date,
    MAX(LIABILITY_START_DATE) AS max_liability_start_date,
    MIN(LIABILITY_END_DATE) AS min_liability_end_date,
    MAX(LIABILITY_END_DATE) AS max_liability_end_date,
    COUNT(CONTRACT_ID) AS filled_contract_id,
    COUNT(CLIENT_ID) AS filled_client_id,
    COUNT(OBJECT_ID) AS filled_object_id,
    COUNT(PRODUCT_GROUP) AS filled_product_group,
    COUNT(PRODUCT_TYPE) AS filled_product_type,
    COUNT(PRODUCT_1C_NAME) AS filled_product_1c_name,
    COUNT(PREM_SUM_RUB) AS filled_prem_sum_rub,
    COUNT(OTV_SUM_RUB) AS filled_otv_sum_rub,
    SUM(CASE WHEN PREM_SUM_RUB IS NULL THEN 1 ELSE 0 END) AS null_prem_sum_rub,
    SUM(CASE WHEN OTV_SUM_RUB IS NULL THEN 1 ELSE 0 END) AS null_otv_sum_rub,
    SUM(CASE WHEN PREM_SUM_RUB = 0 THEN 1 ELSE 0 END) AS zero_prem_sum_rub,
    SUM(CASE WHEN OTV_SUM_RUB = 0 THEN 1 ELSE 0 END) AS zero_otv_sum_rub,
    SUM(CASE WHEN PREM_SUM_RUB < 0 THEN 1 ELSE 0 END) AS negative_prem_sum_rub,
    SUM(CASE WHEN OTV_SUM_RUB < 0 THEN 1 ELSE 0 END) AS negative_otv_sum_rub
FROM (
    SELECT *
    FROM DM_RISK_AVATAR.CONTRACTS
    WHERE ROWNUM <= 500000
);

96_join_contracts_losses_sample.csv

SELECT
    COUNT(*) AS sample_contract_rows,
    COUNT(l.CONTRACT_ID) AS matched_loss_rows,
    COUNT(DISTINCT c.CONTRACT_ID) AS distinct_contracts,
    COUNT(DISTINCT l.CLAIM_ID) AS distinct_claims,
    SUM(CASE WHEN l.CONTRACT_ID IS NOT NULL THEN 1 ELSE 0 END) AS contracts_with_loss_rows
FROM (
    SELECT CONTRACT_ID
    FROM DM_RISK_AVATAR.CONTRACTS
    WHERE CONTRACT_ID IS NOT NULL
      AND ROWNUM <= 100000
) c
LEFT JOIN DM_RISK_AVATAR.LOSSES l
  ON c.CONTRACT_ID = l.CONTRACT_ID;

Сохрани как:
102_join_stg_contracts_object_sample.csv

SELECT
    COUNT(*) AS sample_stg_rows,
    COUNT(c.CONTRACT_ID) AS matched_contract_rows
FROM (
    SELECT CONTRACT_ID, INSURED_OBJ_ID
    FROM DM_RISK_AVATAR.STG_CONTRACT_OBJRISK
    WHERE CONTRACT_ID IS NOT NULL
      AND INSURED_OBJ_ID IS NOT NULL
      AND ROWNUM <= 100000
) s
LEFT JOIN DM_RISK_AVATAR.CONTRACTS c
  ON s.CONTRACT_ID = c.CONTRACT_ID
 AND s.INSURED_OBJ_ID = c.OBJECT_ID;


Сохрани как:
103_join_stg_lnk_object_sample.csv

SELECT
    COUNT(*) AS sample_stg_rows,
    COUNT(l.CONTRACT_ID) AS matched_lnk_rows
FROM (
    SELECT CONTRACT_ID, CLIENT_ID, INSURED_OBJ_ID
    FROM DM_RISK_AVATAR.STG_CONTRACT_OBJRISK
    WHERE CONTRACT_ID IS NOT NULL
      AND CLIENT_ID IS NOT NULL
      AND INSURED_OBJ_ID IS NOT NULL
      AND ROWNUM <= 100000
) s
LEFT JOIN DM_RISK_AVATAR.LNK_CLIENT_OBJECT_CONTRACT l
  ON s.CONTRACT_ID = l.CONTRACT_ID
 AND s.CLIENT_ID = l.CLIENT_ID
 AND s.INSURED_OBJ_ID = l.OBJECT_ID;


Сохрани как:
104_join_lnk_buildings_estate_sample.csv

SELECT
    COUNT(*) AS sample_lnk_rows,
    COUNT(b.CAD_IND) AS matched_buildings_by_cad_ind,
    COUNT(e.CAD_IND) AS matched_estate_by_cad_ind
FROM (
    SELECT OBJECT_ID
    FROM DM_RISK_AVATAR.LNK_CLIENT_OBJECT_CONTRACT
    WHERE OBJECT_ID IS NOT NULL
      AND ROWNUM <= 100000
) l
LEFT JOIN DM_RISK_AVATAR.BUILDINGS b
  ON l.OBJECT_ID = TO_CHAR(b.CAD_IND)
LEFT JOIN DM_RISK_AVATAR.ESTATE_OBJECTS e
  ON l.OBJECT_ID = TO_CHAR(e.CAD_IND);
